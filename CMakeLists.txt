cmake_minimum_required(VERSION 3.20)
project(XAVRX_WII VERSION 0.1.0 LANGUAGES CXX)
option(XAVRX_BUILD_TESTS "Build the test suite" ON)
add_library(xavrx_core src/core/emulator.cpp src/core/broadway_cpu.cpp src/cpu/cpu_state.cpp src/cpu/decoder.cpp src/cpu/executor.cpp src/cpu/memory_access.cpp src/cpu/control_flow.cpp src/memory/memory.cpp src/system/hardware.cpp src/system/wii_memory.cpp src/system/hollywood.cpp src/system/starlet_memory.cpp src/system/wii_system.cpp src/system/peripherals.cpp src/system/drive_interface.cpp src/system/controllers.cpp src/system/input_mapping.cpp src/system/reset_controller.cpp src/system/dsp_gpu.cpp src/system/broadway_devices.cpp)
target_include_directories(xavrx_core PUBLIC include)
target_compile_features(xavrx_core PUBLIC cxx_std_20)
if(MSVC)
 target_compile_options(xavrx_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(xavrx_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(xavrx-wii src/main.cpp)
target_link_libraries(xavrx-wii PRIVATE xavrx_core)
